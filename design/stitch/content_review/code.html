// Extracted Candidate Routine
Node* reverseList(Node* head) {
  Node* prev = nullptr; Node* curr = head;
  while (curr != nullptr) {
    Node* nextTemp = curr->next;
    curr->next = prev; prev = curr; curr = nextTemp;
  }
  return prev;
}
